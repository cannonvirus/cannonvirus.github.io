{{- /* 벤치 리더보드 표. 인자는 벤치 자료에 id를 더한 것이다(partials/bench/boards.html). 평균은 avg가 true인 열의 단순 평균이고, 평균 내림차순으로 줄을 세운다. 블로그 글의 표 스타일을 쓰고, 평균 열과 열마다 가장 좋은 값만 굵게 한다. lower가 true인 열은 낮을수록 좋다. 머리글을 눌러 다시 정렬하는 일은 assets/bench.js가 한다. */ -}} {{- $d := . -}} {{- $cols := $d.columns -}} {{- $avgKeys := slice }}{{ range $cols }}{{ if .avg }}{{ $avgKeys = $avgKeys | append .key }}{{ end }}{{ end -}} {{- $isNum := slice "float64" "int" "int64" -}} {{- /* 열마다 가장 좋은 값 */ -}} {{- $best := dict -}} {{- range $cols -}} {{- $c := . }}{{ $k := .key }}{{ $vals := slice -}} {{- range $d.rows }}{{ $v := index . $k }}{{ if in $isNum (printf "%T" $v) }}{{ $vals = $vals | append $v }}{{ end }}{{ end -}} {{- with $vals }}{{ $best = merge $best (dict $k (cond (default false $c.lower) (math.Min .) (math.Max .))) }}{{ end -}} {{- end -}} {{- /* 줄마다 평균. 표시하는 둘째 자리로 반올림한 값으로 줄을 세워 표시와 순서가 어긋나지 않게 한다. */ -}} {{- $rows := slice -}} {{- range $d.rows -}} {{- $r := . }}{{ $sum := 0.0 -}} {{- range $avgKeys }}{{ $sum = add $sum (float (index $r .)) }}{{ end -}} {{- $avg := div (math.Round (mul (div $sum (len $avgKeys)) 100)) 100.0 -}} {{- $rows = $rows | append (dict "r" $r "avg" $avg) -}} {{- end -}} {{- $rows = sort $rows "avg" "desc" -}} {{- /* 머리 위 묶음 줄. 이웃한 같은 group을 한 칸으로 합친다. */ -}} {{- $groups := slice }}{{ $cur := "" }}{{ $n := 0 }}{{ $any := false -}} {{- range $i, $c := $cols -}} {{- $g := .group | default "" }}{{ if $g }}{{ $any = true }}{{ end -}} {{- if and (gt $i 0) (eq $g $cur) }}{{ $n = add $n 1 -}} {{- else }}{{ if gt $i 0 }}{{ $groups = $groups | append (dict "label" $cur "span" $n) }}{{ end }}{{ $cur = $g }}{{ $n = 1 }}{{ end -}} {{- end -}} {{- $groups = $groups | append (dict "label" $cur "span" $n) -}}
